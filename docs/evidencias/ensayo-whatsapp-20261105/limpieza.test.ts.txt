import {manejarLimpiezaEnsayo,LIMPIEZA} from '../../supabase/functions/_shared/whatsapp-ensayo-limpieza.ts';
const assert=(v:unknown)=>{if(!v)throw new Error('assert');};
function fixture(){const llamadas:string[]=[];const estado={ok:true,cerrado:true,es_prueba:true,estado_registro:'cancelado',ensayo_id:LIMPIEZA.run,registro_id:LIMPIEZA.registro,invitacion_id:LIMPIEZA.invitacion,envio_id:LIMPIEZA.envio,intentos:1,estado_envio:'aceptado_meta'};const d={env:(k:string)=>({WHATSAPP_ENSAYO_PLATAFORMA_HABILITADO:'false',WHATSAPP_ENVIOS_HABILITADOS:'false',WHATSAPP_MODO:'simulacion'}[k]),identidad:async()=> 'admin',rpc:async(_a:string,accion:string)=>{llamadas.push(accion);return accion==='estado'?estado:true;},eliminarMedia:async(id:string)=>{assert(id===LIMPIEZA.media);llamadas.push('media');},eliminarFoto:async(path:string)=>{assert(path===LIMPIEZA.foto);llamadas.push('foto');}};return{d,estado,llamadas};}
const req=(b:unknown={accion:'cerrar',confirmar:true},headers:Record<string,string>={Authorization:'Bearer valid',Origin:'http://127.0.0.1:5173'})=>new Request('http://localhost/',{method:'POST',headers,body:JSON.stringify(b)});
Deno.test('limpieza exacta despues de cierre; media antes de foto',async()=>{const f=fixture();const r=await manejarLimpiezaEnsayo(f.d)(req());assert(r.status===200);assert(f.llamadas.join(',')==='permiso,estado,media,medio_eliminado,foto');});
Deno.test('sin JWT no toca medios',async()=>{const f=fixture();assert((await manejarLimpiezaEnsayo(f.d)(req({},{}))).status===401);assert(!f.llamadas.length);});
Deno.test('JWT invalido no toca medios',async()=>{const f=fixture();f.d.identidad=async()=>null as unknown as string;assert((await manejarLimpiezaEnsayo(f.d)(req())).status===401);assert(!f.llamadas.length);});
Deno.test('origen ajeno bloqueado',async()=>{const f=fixture();assert((await manejarLimpiezaEnsayo(f.d)(req({}, {Origin:'https://evil.example',Authorization:'Bearer valid'}))).status===403);assert(!f.llamadas.length);});
Deno.test('no permite identificadores ni otros comandos',async()=>{for(const b of [{accion:'crear',confirmar:true},{accion:'cerrar',confirmar:false},{accion:'cerrar',confirmar:true,id:'other'}]){const f=fixture();assert((await manejarLimpiezaEnsayo(f.d)(req(b))).status===400);assert(!f.llamadas.length);}});
Deno.test('ensayo activo o cola general encendida bloquean',async()=>{const f=fixture();f.d.env=()=> 'true';assert((await manejarLimpiezaEnsayo(f.d)(req())).status===409);assert(!f.llamadas.length);});
Deno.test('registro real, otro ID o ensayo abierto nunca se borran',async()=>{for(const change of [{es_prueba:false},{cerrado:false},{registro_id:'otro'},{intentos:0}]){const f=fixture();Object.assign(f.estado,change);assert((await manejarLimpiezaEnsayo(f.d)(req())).status===409);assert(!f.llamadas.includes('media'));}});
Deno.test('error Meta impide marcar o borrar foto',async()=>{const f=fixture();f.d.eliminarMedia=async()=>{throw new Error('provider');};assert((await manejarLimpiezaEnsayo(f.d)(req())).status===503);assert(!f.llamadas.includes('foto'));assert(!f.llamadas.includes('medio_eliminado'));});
Deno.test('permiso RPC rechazado impide toda eliminacion',async()=>{const f=fixture();f.d.rpc=async()=>{throw new Error('denied');};assert((await manejarLimpiezaEnsayo(f.d)(req())).status===503);assert(!f.llamadas.length);});
